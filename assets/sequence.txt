; sequence.txt -- game flow: named blocks, each ending in a NEXT step naming the block that follows
; Format:
;   label:id [secret]     block header; id optional; becomes --scene name "labelid"
;     TASK key=val ...    task in the block, run in order (must be indented with spaces or tab)
;     NEXT <label:id>     the block's last line (required, exactly once): the successor block.
;                         NEXT end is the deliberate terminal. NEXT is an exit, not a task.
;   secret marks a block no NEXT reaches: only an explicit jump enters it.
;   tests/seq_graph_test.bas fails on a dangling NEXT, a block with no NEXT, or an unreachable
;   non-secret block.
;
; Task types: CRAWL CARD TITLE PLAY ARRIVE DOCK
; CRAWL  txt=<gtext-key>  mus=<cue>
; CARD   img=<image-key>  mus=<cue>   (EMPEROR is a backward-compat alias)
; TITLE  mus=<cue>
; PLAY   type=combat|boss|asteroid  mus=<cue>  [trigger=<n>]
;   type=boss: mus=cue1,cue2,cue3       -- comma list, one cue per boss phase (tickable)
;              speech=key1,key2,key3  -- comma list of gametext keys (empty entry = no speech)
;              maneuver=name1,name2,name3 -- comma list of flight patterns per phase (assets/maneuvers/*.mvr)
;              seek=N -- boss bullet homing strength per frame (0=straight, 0.05=moderate curve)
;              attack=a,b,... -- the boss's attack pool (#26): laser, spore, cut, devour; one picked at random per volley
;              hp=N -- the boss's hit points (#90; none = boss_hp_default in assets/config/bosses.ini; --nerf: x 0.1)
;              secs=N -- seconds the fight is expected to take, for the dock's NEXT JUMP NEEDS (#90; none = fuel_boss_secs in assets/config/economy.ini)
;   trigger=n: combat: enemy kills that end the stage (nerf: * NERF_FACTOR, at least 1); asteroid: parsecs
; ARRIVE mus=<cue>
; DOCK   [mus=<cue>]   upgrade dock: spend credits between stages (none after the last stage)

crawl:0
    CRAWL txt=intro mus=crawl
    NEXT emperor

emperor:
    CARD img=emperor mus=emperor
    NEXT title

title:
    TITLE mus=title
    NEXT level:1

level:1
    CRAWL txt=stage1 mus=crawl
    PLAY type=combat mus=game trigger=12
    ;PLAY type=boss maneuver=latest_greatest mus=boss,boss2,boss3 speech=,speech_boss_phase2,speech_boss_phase3 seek=0.015
    ARRIVE mus=planet
    DOCK mus=dock
    NEXT level:2

level:2
    CRAWL txt=stage2 mus=crawl
    PLAY type=combat mus=game trigger=18
    ARRIVE mus=planet
    DOCK mus=dock
    NEXT level:3

level:3
    CRAWL txt=stage3 mus=crawl
    PLAY type=combat mus=game trigger=12
    PLAY type=boss maneuver=boss-x-flight,boss-v-flight,attack-pass mus=boss,boss2,boss3 speech=,speech_boss_phase2,speech_boss_phase3 seek=0.015 attack=laser,spore hp=30 secs=22
    ARRIVE mus=planet
    DOCK mus=dock
    NEXT level:4

level:4
    CRAWL txt=stage4 mus=crawl
    PLAY type=asteroid mus=asteroid trigger=350 
    ARRIVE mus=planet
    DOCK mus=dock
    NEXT level:5

level:5
    CRAWL txt=stage5 mus=crawl
    PLAY type=combat mus=game trigger=18
    PLAY type=boss maneuver=boss-x-flight,boss-v-flight,attack-pass mus=boss,boss2,boss3 speech=,speech_boss_phase2,speech_boss_phase3 seek=0.035 attack=laser,spore,cut hp=90 secs=25
    ARRIVE mus=planet
    DOCK mus=dock
    NEXT level:6

level:6
    CRAWL txt=stage6 mus=crawl
    PLAY type=combat mus=game trigger=24
    PLAY type=boss maneuver=boss-x-flight,boss-v-flight,attack-pass mus=boss,boss2,boss3 speech=,speech_boss_phase2,speech_boss_phase3 seek=0.055 attack=laser,spore,cut,devour hp=130 secs=32
    ARRIVE mus=planet
    NEXT outro

outro:
    CRAWL txt=outro mus=crawl
    TITLE mus=title
    NEXT end

; the isolated tutorial run (#240): entered by --scene tutorial, the title menu or the first New Game via
; GAME_StartTutorial, never by a NEXT. script= hands the flight to the step runner (assets/tutorial.txt,
; src/sys/tutorial.bas), which ends the run; the combat kill trigger below is not used.
tutorial: secret
    PLAY type=combat script=tutorial mus=game trigger=3
    TITLE mus=title
    NEXT end
