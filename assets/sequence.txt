; sequence.txt -- game scene order
; Format:
;   label:id              section header; id optional; becomes --scene name "labelid"
;     TASK key=val ...    task under the label (must be indented with spaces or tab)
;
; Task types: CRAWL CARD TITLE PLAY ARRIVE
; CRAWL  txt=<gtext-key>  mus=<cue>
; CARD   img=<image-key>  mus=<cue>   (EMPEROR is a backward-compat alias)
; TITLE  mus=<cue>
; PLAY   type=combat|boss|asteroid  mus=<cue>  [trigger=<n>]
;   type=boss: mus=cue1,cue2,cue3       -- comma list, one cue per boss phase (tickable)
;              speech=key1,key2,key3  -- comma list of gametext keys (empty entry = no speech)
;              maneuver=name1,name2,name3 -- comma list of flight patterns per phase (maneuvers.txt)
;              seek=N -- boss bullet homing strength per frame (0=straight, 0.05=moderate curve)
;   trigger=n: score threshold in units of 100 pts (nerf: * NERF_FACTOR)
; ARRIVE mus=<cue>

crawl:0
    CRAWL txt=intro mus=crawl

emperor:
    CARD img=emperor mus=emperor

title:
    TITLE mus=title

level:1
    CRAWL txt=stage1 mus=crawl
    PLAY type=combat mus=game trigger=10
    ARRIVE mus=planet

level:2
    CRAWL txt=stage2 mus=crawl
    PLAY type=combat mus=game trigger=15
    ARRIVE mus=planet

level:3
    CRAWL txt=stage3 mus=crawl
    PLAY type=combat mus=game trigger=10
    PLAY type=boss maneuver=drift mus=boss,boss2,boss3 speech=,speech_boss_phase2,speech_boss_phase3 seek=0.015
    ARRIVE mus=planet

level:4
    CRAWL txt=stage4 mus=crawl
    PLAY type=asteroid mus=asteroid trigger=350 
    ARRIVE mus=planet

level:5
    CRAWL txt=stage5 mus=crawl
    PLAY type=combat mus=game trigger=15
    PLAY type=boss maneuver=drift,weave mus=boss,boss2,boss3 speech=,speech_boss_phase2,speech_boss_phase3 seek=0.035
    ARRIVE mus=planet

level:6
    CRAWL txt=stage6 mus=crawl
    PLAY type=combat mus=game trigger=20
    PLAY type=boss maneuver=weave,flyover,plotted mus=boss,boss2,boss3 speech=,speech_boss_phase2,speech_boss_phase3 seek=0.055
    ARRIVE mus=planet

outro:
    CRAWL txt=outro mus=crawl
    TITLE mus=title
