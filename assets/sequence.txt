; sequence.txt -- game scene order
; Format:
;   label:id              section header; id optional; becomes --scene name "labelid"
;     TASK key=val ...    task under the label (must be indented with spaces or tab)
;
; Task types: CRAWL CARD TITLE PLAY ARRIVE DOCK
; CRAWL  txt=<gtext-key>  mus=<cue>
; CARD   img=<image-key>  mus=<cue>   (EMPEROR is a backward-compat alias)
; TITLE  mus=<cue>
; PLAY   type=combat|boss|asteroid  mus=<cue>  [trigger=<n>]
;   type=boss: mus=cue1,cue2,cue3       -- comma list, one cue per boss phase (tickable)
;              speech=key1,key2,key3  -- comma list of gametext keys (empty entry = no speech)
;              maneuver=name1,name2,name3 -- comma list of flight patterns per phase (assets/maneuvers/*.mvr)
;              seek=N -- boss bullet homing strength per frame (0=straight, 0.05=moderate curve)
;              attack=a,b,... -- the boss's attack pool (#26): laser, spore, cut, devour; one picked at random per volley
;              hp=N -- the boss's hit points (#90; none = boss_hp_default in assets/config/bosses.ini; --nerf: x 0.1)
;   trigger=n: combat: enemy kills that end the stage (nerf: * NERF_FACTOR, at least 1); asteroid: parsecs
; ARRIVE mus=<cue>
; DOCK   [mus=<cue>]   upgrade dock: spend credits between stages (none after the last stage)

crawl:0
    CRAWL txt=intro mus=crawl

emperor:
    CARD img=emperor mus=emperor

    TITLE mus=title

    ARRIVE mus=planet
