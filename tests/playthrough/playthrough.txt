# Full-playthrough input script for tests/playthrough/run.sh (#100), played with
#! seed 1
#! args --god --nerf
#! frames 13100
# from a launch folder with no sss_settings.ini. The outro and credits return to the title at
# about frame 12700; run.sh checks the milestones against
# expected.txt. Every line reacts to the screen it is on (@ lines, #328), taking each
# screen's quickest way out, so no screen waits on a retry.
#
# Data notice: SPACE is OK.
@GS_CONSENT +0 SPACE
# Callsign "ace7", then ENTER (one typed key per frame).
@GS_USERNAME +0 A
@GS_USERNAME +1 C
@GS_USERNAME +2 E
@GS_USERNAME +3 1 7
@GS_USERNAME +4 ENTER
# Lead-in cards: any key press skips one.
@GS_LEADIN +0 SPACE
@GS_LEADIN +1 SPACE
@GS_LEADIN +2 SPACE
# The intro crawl fast-forwards (FIRE) for 240 frames before the skip, so its first
# paragraphs start speaking: tests/playthrough/determinism.sh compares those NARR lines.
@GS_CRAWL#1 +60 240 FIRE
@GS_CRAWL#1 +300 FIRE ENTER
# Every other crawl: ENTER (with FIRE held) skips it once it is 60 frames old.
@GS_CRAWL#2 +60 FIRE ENTER
@GS_CRAWL#3 +60 FIRE ENTER
@GS_CRAWL#4 +60 FIRE ENTER
@GS_CRAWL#5 +60 FIRE ENTER
@GS_CRAWL#6 +60 FIRE ENTER
@GS_CRAWL#7 +60 FIRE ENTER
@GS_CRAWL#8 +60 FIRE ENTER
# The credits crawl follows the outro with no state change, so it is skipped by age from the outro's entry.
@GS_CRAWL#8 +130 FIRE ENTER
# Emperor card: ESC goes back to the title (the crawl skip held ENTER, not ESC).
@GS_INTRO +1 ESC
# Title: P opens LAUNCH NOW?, a second P five frames later launches. Only the first time;
# the outro's return to the title ends the run.
@GS_TITLE#1 +0 P
@GS_TITLE#1 +5 P
# Flight: fire throughout, and weave LEFT, UP, RIGHT, DOWN (60 frames each) so aim
# assist finds targets and the asteroid field gets crossed.
@GS_PLAYING +0 * FIRE
@GS_PLAYING +0 60 every 240 LEFT
@GS_PLAYING +60 60 every 240 UP
@GS_PLAYING +120 60 every 240 RIGHT
@GS_PLAYING +180 60 every 240 DOWN
# Planet arrivals and cinematics play out by themselves.
# Upgrade docks (after stages 1-5): L launches, buying nothing. Two frames in, so the key is seen going down (the dock ignores an L already held when it opens).
@GS_DOCK +2 L
