# Full-playthrough input script for tests/playthrough/run.sh (#100), played with
#! seed 1
#! args --god --nerf
#! frames 12900
# from a launch folder with no sss_settings.ini. The outro returns to the title at
# frame 12377; run.sh checks the milestones against
# expected.txt. Every line reacts to the screen it is on (@ lines, #328), taking each
# screen's quickest way out, so no screen waits on a retry.
#
# Data notice: SPACE is OK.
@GS_CONSENT +0 SPACE
# Callsign "ace7", then ENTER (one typed key per frame).
@GS_USERNAME +0 A
@GS_USERNAME +1 C
@GS_USERNAME +2 E
@GS_USERNAME +3 1 7
@GS_USERNAME +4 ENTER
# Lead-in cards: any key press skips one.
@GS_LEADIN +0 SPACE
@GS_LEADIN +1 SPACE
@GS_LEADIN +2 SPACE
# Every crawl: SPACE + ESC skips it once it is 60 frames old.
@GS_CRAWL +60 FIRE ESC
# Emperor card: a fresh ESC skips to the title. The crawl skip just held ESC, so it
# waits a frame for the card to see ESC up first.
@GS_INTRO +1 ESC
# Title: P plays. Only the first time; the outro's return to the title ends the run.
@GS_TITLE#1 +0 P
# Flight: fire throughout, and weave LEFT, UP, RIGHT, DOWN (60 frames each) so aim
# assist finds targets and the asteroid field gets crossed.
@GS_PLAYING +0 * FIRE
@GS_PLAYING +0 60 every 240 LEFT
@GS_PLAYING +60 60 every 240 UP
@GS_PLAYING +120 60 every 240 RIGHT
@GS_PLAYING +180 60 every 240 DOWN
# Planet arrivals and cinematics play out by themselves.
# Upgrade docks (after stages 1-5): ESC leaves at once, buying nothing.
@GS_DOCK +0 ESC
